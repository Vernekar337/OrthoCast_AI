import axios from 'axios';

// Base URL resolution: honors VITE_API_BASE_URL, falls back to relative /api or localhost
const resolveBaseUrl = () => {
  const envUrl = import.meta.env.VITE_API_BASE_URL;
  if (envUrl) {
    // Strip trailing slash if present
    return envUrl.replace(/\/+$/, '');
  }
  return '/api/v1';
};

export const API_BASE_URL = resolveBaseUrl();

// Centralized Axios instance
export const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 30000, // 30s standard timeout for clinical API
  headers: {
    'Content-Type': 'application/json',
    'Accept': 'application/json',
  },
});

/**
 * Standardized API Error format
 */
export class ApiClientError extends Error {
  constructor(status, code, message, hint = null, raw = null) {
    super(message);
    this.name = 'ApiClientError';
    this.status = status;
    this.code = code;
    this.hint = hint;
    this.raw = raw;
  }
}

// Response interceptor to normalize error responses
apiClient.interceptors.response.use(
  (response) => response.data,
  (error) => {
    // Handle offline status
    if (typeof window !== 'undefined' && !window.navigator.onLine) {
      throw new ApiClientError(
        0,
        'offline',
        'Device is offline. Check clinical network connectivity.',
        'Reconnect to hospital or local network.'
      );
    }

    if (error.response) {
      const { status, data } = error.response;
      // FastAPI ApiError format: { detail: { code, message, hint } } or { detail: "message" }
      let message = 'An unexpected server error occurred.';
      let code = 'server_error';
      let hint = null;

      if (typeof data?.detail === 'string') {
        message = data.detail;
        code = `http_${status}`;
      } else if (data?.detail && typeof data.detail === 'object') {
        if (Array.isArray(data.detail)) {
          // FastAPI Pydantic validation error format
          code = 'validation_error';
          message = data.detail.map((err) => `${err.loc?.slice(1).join('.')}: ${err.msg}`).join(', ');
          hint = 'Please check the entered values match clinical parameter boundaries.';
        } else {
          message = data.detail.message || data.detail.msg || message;
          code = data.detail.code || code;
          hint = data.detail.hint || null;
        }
      } else if (data?.message) {
        message = data.message;
        code = data.code || `http_${status}`;
        hint = data.hint || null;
      }

      throw new ApiClientError(status, code, message, hint, data);
    }

    if (error.code === 'ECONNABORTED' || error.message.includes('timeout')) {
      throw new ApiClientError(
        408,
        'timeout',
        'Request timed out. Pipeline or CAD server took too long to respond.',
        'Check backend processing load or try again.'
      );
    }

    // Network error / server unreachable
    throw new ApiClientError(
      0,
      'network_error',
      'Unable to connect to OrthoCast backend service.',
      `Check if FastAPI server is running on ${API_BASE_URL} and MongoDB service is active.`,
      error
    );
  }
);

/**
 * Helper for multipart form upload with upload progress
 */
export async function uploadMultipart(url, formData, onUploadProgress) {
  return apiClient.post(url, formData, {
    headers: {
      // Allow browser and Axios to automatically compute Content-Type with boundary
      'Content-Type': undefined,
    },
    timeout: 120000, // 2 minutes for high-res photo/DICOM/scan uploads
    onUploadProgress: (progressEvent) => {
      if (onUploadProgress && progressEvent.total) {
        const percent = Math.round((progressEvent.loaded * 100) / progressEvent.total);
        onUploadProgress(percent, progressEvent.loaded, progressEvent.total);
      }
    },
  });
}

/**
 * Server-Sent Events / Stream reader for pipeline job tracking
 */
export function streamJobProgress(jobId, onEvent, onError, onComplete) {
  let active = true;
  let seenSeq = 0;

  const poll = async () => {
    if (!active) return;
    try {
      // 1. Fetch latest events list
      try {
        const eventsRes = await apiClient.get(`/jobs/${jobId}/events`);
        const eventList = eventsRes?.events || [];
        for (const evt of eventList) {
          if (evt.seq > seenSeq) {
            seenSeq = evt.seq;
            if (onEvent) onEvent(evt);
          }
        }
      } catch (evtErr) {
        // Continue to main status check
      }

      // 2. Fetch main job document
      const job = await apiClient.get(`/jobs/${jobId}`);
      if (!active) return;
      if (onEvent && seenSeq === 0) onEvent(job);

      if (job.status === 'completed') {
        active = false;
        if (onComplete) onComplete(job);
      } else if (job.status === 'failed') {
        active = false;
        const errDetail = (typeof job.error === 'object' && job.error?.message) ? job.error.message : (job.error || 'Job processing failed');
        if (onError) onError(new ApiClientError(500, 'job_failed', errDetail));
      }
    } catch (e) {
      // Transient network retry
    }
  };

  poll();
  const interval = setInterval(poll, 1500);

  return () => {
    active = false;
    clearInterval(interval);
  };
}

export default apiClient;

import { apiClient } from './client';

export async function getPrediction(patientId) {
  try {
    const response = await apiClient.get(`/patients/${patientId}`);
    const predictions = response?.predictions ?? [];
    const rec = predictions[0] ?? null;
    if (!rec) return null;
    const inner = (typeof rec.data === 'object' && rec.data !== null) ? rec.data : (rec.prediction || {});
    return {
      ...inner,
      id: rec.id || rec._id,
      patient_id: rec.patient_id,
      measurement_id: rec.measurement_id,
      region: rec.region || inner.region,
      created_at: rec.created_at,
      data: inner,
    };
  } catch (err) {
    return null;
  }
}

export async function runPrediction(patientId) {
  const res = await apiClient.post(`/patients/${patientId}/predictions`);
  const inner = (typeof res?.data === 'object' && res?.data !== null) ? res.data : (res?.prediction || res || {});
  return {
    ...inner,
    id: res?.id || res?._id || res?.prediction_id,
    patient_id: res?.patient_id || patientId,
    measurement_id: res?.measurement_id,
    created_at: res?.created_at,
    data: inner,
  };
}

import React from 'react';
import { Loader2 } from 'lucide-react';

/**
 * Reusable Button with loading state and clinical styling
 */
export default function Button({
  children,
  variant = 'primary', // 'primary' | 'teal' | 'secondary' | 'outline' | 'ghost' | 'danger'
  size = 'md', // 'sm' | 'md' | 'lg'
  isLoading = false,
  icon: Icon = null,
  disabled = false,
  className = '',
  type = 'button',
  onClick = null,
  ...props
}) {
  const variantClass = {
    primary:   'btn-primary',
    teal:      'btn-teal',
    secondary: 'btn-secondary',
    outline:   'btn-outline',
    ghost:     'btn-ghost',
    danger:    'btn-danger',
  }[variant] || 'btn-primary';

  const sizeClass = {
    sm: 'btn-sm',
    md: '',
    lg: 'btn-lg',
  }[size] || '';

  return (
    <button
      type={type}
      disabled={disabled || isLoading}
      onClick={onClick}
      className={`btn ${variantClass} ${sizeClass} ${className}`}
      {...props}
    >
      {isLoading ? (
        <Loader2 className="w-3.5 h-3.5 animate-spin flex-shrink-0" />
      ) : Icon ? (
        <Icon className="w-3.5 h-3.5 flex-shrink-0" />
      ) : null}
      <span>{children}</span>
    </button>
  );
}

import React from 'react';
import { AlertTriangle, RefreshCw, WifiOff, HelpCircle } from 'lucide-react';

/**
 * Diagnostic error display for clinical engineering failures
 */
export default function ErrorState({
  error,
  title = 'System Error',
  onRetry = null,
  compact = false,
}) {
  const isOffline = error?.code === 'offline' || (typeof window !== 'undefined' && !window.navigator.onLine);
  const message = error?.message || (typeof error === 'string' ? error : 'An unexpected error occurred.');
  const code = error?.code || 'UNKNOWN_ERROR';
  const hint = error?.hint || null;

  if (compact) {
    return (
      <div className="flex items-center justify-between gap-3 p-3 rounded-md bg-rose-950/30 border border-rose-800/40 text-xs">
        <div className="flex items-center gap-2 text-rose-300">
          <AlertTriangle className="w-4 h-4 flex-shrink-0 text-rose-400" />
          <span className="truncate">{message}</span>
        </div>
        {onRetry && (
          <button
            onClick={onRetry}
            className="btn btn-outline btn-sm text-[10px] text-rose-300 border-rose-700/50 hover:bg-rose-900/30 flex-shrink-0"
          >
            Retry
          </button>
        )}
      </div>
    );
  }

  return (
    <div className="p-6 rounded-lg bg-rose-950/20 border border-rose-800/40 text-center max-w-lg mx-auto">
      <div className="w-12 h-12 rounded-full bg-rose-900/30 border border-rose-700/50 flex items-center justify-center mx-auto mb-3 text-rose-400">
        {isOffline ? <WifiOff className="w-6 h-6" /> : <AlertTriangle className="w-6 h-6" />}
      </div>

      <div className="inline-block px-2 py-0.5 rounded bg-rose-900/40 text-[10px] font-mono uppercase tracking-wider text-rose-300 mb-2 border border-rose-700/30">
        Diagnostic Code: {code}

        {title}
      </h3>

      <p className="text-xs text-rose-300/90 mt-1 leading-relaxed">
        {message}
      </p>

      {hint && (
        <div className="mt-3 p-2.5 rounded bg-slate-900/60 border border-slate-800 text-[11px] text-slate-300 flex items-start gap-2 text-left">

          <div>
            <span className="font-semibold text-cyan-300">Remediation: </span>

          </div>
        </div>
      )}

          <RefreshCw className="w-3 h-3" />
          Retry Operation
        </button>
      )}

  );

/**
 * Procedural mesh generator reproducing server/Cast_logic/generate_cast.py lofting
 * Used as parametric live preview when raw STL file is not yet generated
 */
function createParametricCastMeshes(params, viewMode = 'assembled', explodedDist = 30) {
  const {
    station_diameters_mm = [58.2, 71.4, 84.0],
    total_length_mm = 180,
    wall_thickness_mm = 4.2,
    seam_angle_deg = 0,
    vent = { enabled: true, pattern: 'staggered', hole_mm: 6.0, spacing_mm: 18.0 },
    tabs = { count: 3 },
  } = params;

  const nPts = 48;
  const nStations = station_diameters_mm.length;
  const length = Math.max(60, total_length_mm);
  const wall = Math.max(2.0, wall_thickness_mm);

  // Generate Half A (0 to PI) and Half B (PI to 2*PI)
  const group = new THREE.Group();

  const createHalfGeometry = (thetaStart, thetaLength, isHalfA) => {
    const geometry = new THREE.BufferGeometry();
    const vertices = [];
    const indices = [];
    const normals = [];

    // Outer & Inner rings
    const zStep = length / (nStations - 1);
    const nRings = nStations;
    const radialPts = nPts / 2;

    for (let r = 0; r < nRings; r++) {
      const z = -length / 2 + r * zStep;
      const outerRad = station_diameters_mm[r] / 2;
      const innerRad = Math.max(10, outerRad - wall);

      // Outer ring points
      for (let i = 0; i <= radialPts; i++) {
        const theta = thetaStart + (i / radialPts) * thetaLength;
        const x = outerRad * Math.cos(theta);
        const y = outerRad * Math.sin(theta);
        vertices.push(x, y, z);
        normals.push(Math.cos(theta), Math.sin(theta), 0);
      }

      // Inner ring points
      for (let i = 0; i <= radialPts; i++) {
        const theta = thetaStart + (i / radialPts) * thetaLength;
        const x = innerRad * Math.cos(theta);
        const y = innerRad * Math.sin(theta);
        vertices.push(x, y, z);
        normals.push(-Math.cos(theta), -Math.sin(theta), 0);
      }
    }

    const ringStride = (radialPts + 1) * 2;

    // Build faces along the cylinder height
    for (let r = 0; r < nRings - 1; r++) {
      const r0 = r * ringStride;
      const r1 = (r + 1) * ringStride;

      // Outer surface quads
      for (let i = 0; i < radialPts; i++) {
        const a = r0 + i;
        const b = r1 + i;
        const c = r1 + i + 1;
        const d = r0 + i + 1;
        indices.push(a, b, c);
        indices.push(a, c, d);
      }

      // Inner surface quads
      const inOffset = radialPts + 1;
      for (let i = 0; i < radialPts; i++) {
        const a = r0 + inOffset + i;
        const b = r0 + inOffset + i + 1;
        const c = r1 + inOffset + i + 1;
        const d = r1 + inOffset + i;
        indices.push(a, b, c);
        indices.push(a, c, d);
      }

      // Seam edge caps along the split plane (start edge)
      const aStartOuter0 = r0;
      const aStartInner0 = r0 + inOffset;
      const aStartOuter1 = r1;
      const aStartInner1 = r1 + inOffset;
      indices.push(aStartOuter0, aStartInner0, aStartInner1);
      indices.push(aStartOuter0, aStartInner1, aStartOuter1);

      // Seam edge caps along the split plane (end edge)
      const aEndOuter0 = r0 + radialPts;
      const aEndInner0 = r0 + inOffset + radialPts;
      const aEndOuter1 = r1 + radialPts;
      const aEndInner1 = r1 + inOffset + radialPts;
      indices.push(aEndOuter0, aEndOuter1, aEndInner1);
      indices.push(aEndOuter0, aEndInner1, aEndInner0);
    }

    // Rim caps (bottom rim)
    const bOuter = 0;
    const bInner = radialPts + 1;
    for (let i = 0; i < radialPts; i++) {
      indices.push(bOuter + i, bOuter + i + 1, bInner + i + 1);
      indices.push(bOuter + i, bInner + i + 1, bInner + i);
    }

    // Rim caps (top rim)
    const tBase = (nRings - 1) * ringStride;
    const tOuter = tBase;
    const tInner = tBase + radialPts + 1;
    for (let i = 0; i < radialPts; i++) {
      indices.push(tOuter + i, tInner + i + 1, tOuter + i + 1);
      indices.push(tOuter + i, tInner + i, tInner + i + 1);
    }

    geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));
    geometry.setIndex(indices);
    geometry.computeVertexNormals();
    return geometry;
  };

  const seamRad = (seam_angle_deg * Math.PI) / 180;
  const geomA = createHalfGeometry(seamRad, Math.PI, true);
  const geomB = createHalfGeometry(seamRad + Math.PI, Math.PI, false);

  return { geomA, geomB };
}

export default function CastViewport({
  params,
  stlUrls = null, // { halfA: url, halfB: url }
  isGenerating = false,
  modelStatus = 'preview', // 'preview' | 'generated' | 'loading'
  metadata = null,
}) {
  const mountRef = useRef(null);
  const sceneRef = useRef(null);
  const rendererRef = useRef(null);
  const cameraRef = useRef(null);
  const controlsRef = useRef(null);
  const meshesRef = useRef({ meshA: null, meshB: null, group: null });

  // Display modes
  const [viewMode, setViewMode] = useState('assembled'); // 'assembled' | 'half_a' | 'half_b' | 'exploded'
  const [renderStyle, setRenderStyle] = useState('solid'); // 'solid' | 'wireframe' | 'transparent'
  const [showGrid, setShowGrid] = useState(true);
  const [explodedDistance, setExplodedDistance] = useState(35); // mm

  // Camera reset
  const handleResetCamera = useCallback(() => {
    if (!cameraRef.current || !controlsRef.current) return;
    cameraRef.current.position.set(220, 150, 260);
    controlsRef.current.target.set(0, 0, 0);
    controlsRef.current.update();
  }, []);

  const handleFitToView = useCallback(() => {
    if (!cameraRef.current || !controlsRef.current || !meshesRef.current.group) return;
    const box = new THREE.Box3().setFromObject(meshesRef.current.group);
    const size = box.getSize(new THREE.Vector3());
    const center = box.getCenter(new THREE.Vector3());
    const maxDim = Math.max(size.x, size.y, size.z);
    const fov = cameraRef.current.fov * (Math.PI / 180);
    let cameraZ = Math.abs(maxDim / 2 / Math.tan(fov / 2)) * 1.6;
    cameraRef.current.position.set(cameraZ * 0.8, cameraZ * 0.5, cameraZ);
    controlsRef.current.target.copy(center);
    controlsRef.current.update();
  }, []);

  // Initialize Three.js scene
  useEffect(() => {
    const container = mountRef.current;
    if (!container) return;

    const width = container.clientWidth || 600;
    const height = container.clientHeight || 500;

    // Scene
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x070c18);
    sceneRef.current = scene;

    // Camera
    const camera = new THREE.PerspectiveCamera(45, width / height, 1, 3000);
    camera.position.set(220, 160, 260);
    cameraRef.current = camera;

    // Renderer
    const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true, powerPreference: 'high-performance' });
    renderer.setSize(width, height);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.shadowMap.enabled = true;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.2;
    rendererRef.current = renderer;
    container.appendChild(renderer.domElement);

    // OrbitControls
    const controls = new OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;
    controls.dampingFactor = 0.05;
    controls.minDistance = 60;
    controls.maxDistance = 1200;
    controls.target.set(0, 0, 0);
    controlsRef.current = controls;

    // Lighting (Medical Engineering studio setup)
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.9);
    scene.add(ambientLight);

    const dirLight1 = new THREE.DirectionalLight(0x38bdf8, 2.2); // Key cyan light
    dirLight1.position.set(200, 350, 200);
    scene.add(dirLight1);

    const dirLight2 = new THREE.DirectionalLight(0x2dd4bf, 1.4); // Teal fill light
    dirLight2.position.set(-200, -100, -200);
    scene.add(dirLight2);

    const rimLight = new THREE.DirectionalLight(0xffffff, 1.2);
    rimLight.position.set(0, 200, -300);
    scene.add(rimLight);

    // Grid & Axes
    const grid = new THREE.GridHelper(400, 40, 0x0ea5e9, 0x1e293b);
    grid.position.y = -100;
    grid.name = 'gridHelper';
    scene.add(grid);

    const axes = new THREE.AxesHelper(60);
    axes.position.set(-160, -99, -160);
    scene.add(axes);

    // Group for cast model
    const castGroup = new THREE.Group();
    castGroup.name = 'castModelGroup';
    scene.add(castGroup);
    meshesRef.current.group = castGroup;

    // Animation Loop
    let animationFrameId;
    const animate = () => {
      animationFrameId = requestAnimationFrame(animate);
      controls.update();
      renderer.render(scene, camera);
    };
    animate();

    // Resize Observer
    const resizeObserver = new ResizeObserver((entries) => {
      for (let entry of entries) {
        const { width: w, height: h } = entry.contentRect;
        if (w > 0 && h > 0 && cameraRef.current && rendererRef.current) {
          cameraRef.current.aspect = w / h;
          cameraRef.current.updateProjectionMatrix();
          rendererRef.current.setSize(w, h);
        }
      }
    });
    resizeObserver.observe(container);

    // Cleanup
    return () => {
      cancelAnimationFrame(animationFrameId);
      resizeObserver.disconnect();
      controls.dispose();
      renderer.dispose();
      if (renderer.domElement && container.contains(renderer.domElement)) {
        container.removeChild(renderer.domElement);
      }
      scene.clear();
    };
  }, []);

  // Update Grid visibility
  useEffect(() => {
    if (!sceneRef.current) return;
    const grid = sceneRef.current.getObjectByName('gridHelper');
    if (grid) grid.visible = showGrid;
  }, [showGrid]);

  // Build / Load Cast Geometry
  useEffect(() => {
    const castGroup = meshesRef.current.group;
    if (!castGroup || !params) return;

    // Clear previous children
    while (castGroup.children.length > 0) {
      const child = castGroup.children[0];
      if (child.geometry) child.geometry.dispose();
      if (child.material) {
        if (Array.isArray(child.material)) child.material.forEach((m) => m.dispose());
        else child.material.dispose();
      }
      castGroup.remove(child);
    }

    // Material definitions
    const matA = new THREE.MeshPhysicalMaterial({
      color: 0x0ea5e9, // Cyan anterior polymer
      roughness: 0.25,
      metalness: 0.1,
      clearcoat: 0.6,
      clearcoatRoughness: 0.1,
      wireframe: renderStyle === 'wireframe',
      transparent: renderStyle === 'transparent',
      opacity: renderStyle === 'transparent' ? 0.45 : 1.0,
      side: THREE.DoubleSide,
    });

    const matB = new THREE.MeshPhysicalMaterial({
      color: 0x14b8a6, // Teal posterior polymer
      roughness: 0.25,
      metalness: 0.1,
      clearcoat: 0.6,
      clearcoatRoughness: 0.1,
      wireframe: renderStyle === 'wireframe',
      transparent: renderStyle === 'transparent',
      opacity: renderStyle === 'transparent' ? 0.45 : 1.0,
      side: THREE.DoubleSide,
    });

    // Check if raw STL file URLs exist
    if (stlUrls?.halfA && stlUrls?.halfB) {
      const loader = new STLLoader();
      loader.load(
        stlUrls.halfA,
        (geom) => {
          geom.computeVertexNormals();
          const meshA = new THREE.Mesh(geom, matA);
          meshA.name = 'halfA_stl';
          castGroup.add(meshA);
          meshesRef.current.meshA = meshA;
          applyViewTransforms();
        },
        undefined,
        () => {
          // Fallback to procedural mesh if file download fails
          buildProcedural();
        }
      );

      loader.load(
        stlUrls.halfB,
        (geom) => {
          geom.computeVertexNormals();
          const meshB = new THREE.Mesh(geom, matB);
          meshB.name = 'halfB_stl';
          castGroup.add(meshB);
          meshesRef.current.meshB = meshB;
          applyViewTransforms();
        },
        undefined,
        () => {
          buildProcedural();
        }
      );
    } else {
      buildProcedural();
    }

    function buildProcedural() {
      const { geomA, geomB } = createParametricCastMeshes(params, viewMode, explodedDistance);
      const meshA = new THREE.Mesh(geomA, matA);
      meshA.name = 'halfA_mesh';
      const meshB = new THREE.Mesh(geomB, matB);
      meshB.name = 'halfB_mesh';

      castGroup.add(meshA);
      castGroup.add(meshB);
      meshesRef.current.meshA = meshA;
      meshesRef.current.meshB = meshB;

      applyViewTransforms();
    }

    function applyViewTransforms() {
      const { meshA, meshB } = meshesRef.current;
      if (!meshA || !meshB) return;

      meshA.visible = viewMode !== 'half_b';
      meshB.visible = viewMode !== 'half_a';

      if (viewMode === 'exploded') {
        meshA.position.set(explodedDistance, 0, 0);
        meshB.position.set(-explodedDistance, 0, 0);
      } else {
        meshA.position.set(0, 0, 0);
        meshB.position.set(0, 0, 0);
      }
    }
  }, [params, stlUrls, viewMode, renderStyle, explodedDistance]);

  return (
    <div className="relative w-full h-[540px] rounded-lg border border-cyan-500/30 overflow-hidden bg-[#070c18] flex flex-col select-none">
      {/* â”€â”€ Top HUD Control Bar â”€â”€ */}
      <div className="absolute top-3 left-3 right-3 z-10 flex items-center justify-between pointer-events-none flex-wrap gap-2">
        {/* Left: View Mode Toggles */}
        <div className="flex items-center gap-1 p-1 rounded-md bg-slate-900/90 border border-slate-800 backdrop-blur pointer-events-auto">
          {[
            { id: 'assembled', label: 'Assembled', icon: Box },
            { id: 'half_a', label: 'Half A (Anterior)', icon: Eye },
            { id: 'half_b', label: 'Half B (Posterior)', icon: Eye },
            { id: 'exploded', label: 'Exploded View', icon: SplitSquareVertical },
          ].map((mode) => {
            const Icon = mode.icon;
            const isActive = viewMode === mode.id;
            return (
              <button
                key={mode.id}
                onClick={() => setViewMode(mode.id)}
                className={`flex items-center gap-1.5 px-2.5 py-1 rounded text-[11px] font-mono uppercase transition-all ${
                  isActive
                    ? 'bg-cyan-500 text-white font-bold shadow'
                    : 'text-slate-400 hover:text-white hover:bg-slate-800'

                <span className="hidden sm:inline">{mode.label}</span>

        {/* Right: Render Shading & Camera Reset */}
        <div className="flex items-center gap-1.5 p-1 rounded-md bg-slate-900/90 border border-slate-800 backdrop-blur pointer-events-auto">
          {/* Shading mode */}

      } catch (e) {

      }

      } catch (verErr) {

    } finally {

      setSavedParams(params);

export default function PatientDetailPage() {
  const { patientId } = useParams();
  const navigate = useNavigate();

  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  // Loaded sub-entities
  const [patient, setPatient] = useState(null);
  const [scans, setScans] = useState([]);
  const [measurements, setMeasurements] = useState(null);
  const [prediction, setPrediction] = useState(null);
  const [cast, setCast] = useState(null);

  // Edit Modal State
  const [isEditOpen, setIsEditOpen] = useState(false);
  const [editFormData, setEditFormData] = useState({
    name: '',
    age: '',
    gender: 'M',
    height_cm: '',
    injury_region: 'wrist',
    swelling_stage: 'acute',
    notes: '',
  });
  const [isSubmittingEdit, setIsSubmittingEdit] = useState(false);
  const [editError, setEditError] = useState(null);

  // Delete Modal State
  const [isDeleteOpen, setIsDeleteOpen] = useState(false);
  const [isDeleting, setIsDeleting] = useState(false);

  // Primary fetch handler
  const loadPatientData = useCallback(async () => {
    if (!patientId) {
      setError({ message: 'Patient identifier is missing from route.' });
      setLoading(false);
      return;
    }

    setLoading(true);
    setError(null);

    try {
      // 1. Fetch core patient
      const p = await getPatient(patientId);
      if (!p) {
        throw { message: `Patient with ID "${patientId}" was not found in the registry.` };
      }
      setPatient(p);
      setEditFormData({
        name: p.name || '',
        age: p.age || '',
        gender: p.gender || 'M',
        height_cm: p.height_cm || '',
        injury_region: p.injury_region || 'wrist',
        swelling_stage: p.swelling_stage || 'acute',
        notes: p.notes || '',
      });

      // 2. Fetch related sub-records in parallel
      const [scansRes, measRes, predRes, castRes] = await Promise.allSettled([
        getScans(patientId),
        getMeasurements(patientId),
        getPrediction(patientId),
        getCast(patientId),
      ]);

      if (scansRes.status === 'fulfilled' && scansRes.value) {
        setScans(Array.isArray(scansRes.value) ? scansRes.value : scansRes.value.scans || []);
      }
      if (measRes.status === 'fulfilled' && measRes.value) {
        const measurementRecords = Array.isArray(measRes.value)
          ? measRes.value
          : [];

        setMeasurements(
          measurementRecords[0]?.data ?? measurementRecords[0] ?? null
        );;
      }
      if (predRes.status === 'fulfilled' && predRes.value) {
        setPrediction(predRes.value);
      }
      if (castRes.status === 'fulfilled' && castRes.value) {
        setCast(castRes.value);
      }
    } catch (err) {
      setError(err);
    } finally {
      setLoading(false);
    }
  }, [patientId]);

  useEffect(() => {
    loadPatientData();
  }, [loadPatientData]);

  // Handle Edit Submit

    e.preventDefault();
    setIsSubmittingEdit(true);
    setEditError(null);

    try {
      const patchData = {
        name: editFormData.name.trim(),
        age: parseInt(editFormData.age, 10),
        gender: editFormData.gender,
        height_cm: editFormData.height_cm ? parseFloat(editFormData.height_cm) : null,
        injury_region: editFormData.injury_region,
        swelling_stage: editFormData.swelling_stage,
        notes: editFormData.notes ? editFormData.notes.trim() : null,
      };

      await updatePatient(patientId, patchData);
      setIsEditOpen(false);
      await loadPatientData();
    } catch (err) {
      setEditError(err);
    } finally {
      setIsSubmittingEdit(false);
    }
  };

        if (list.length > 0 && !selectedPatientId) {
          setSelectedPatientId(list[0].id || list[0]._id);
        }
      } catch (e) {
        // Ignored
      }
      setLoading(false);
      return;
    }

    setLoading(true);
    setError(null);

    try {
      const [patientRes, measRes, predRes, scansRes, castRes] = await Promise.allSettled([
        getPatient(selectedPatientId),
        getMeasurements(selectedPatientId),

        getScans(selectedPatientId),
        getCast(selectedPatientId),

      if (patientRes.status === 'fulfilled' && patientRes.value) {

      let status = 'pass'; // 'pass' | 'warning' | 'fail'

      if (!isNaN(manual) && !isNaN(system) && manual > 0) {

        if (pctError > 10.0 || delta > 8.0) {

        } else if (pctError > 5.0 || delta > 4.0) {
          status = 'warning';
        } else {

        <Route path="*" element={<Navigate to="/dashboard" replace />} />
      </Routes>
    </BrowserRouter>
  );

## 2. Detailed Bug Findings & Resolutions

### BUG-001: URL Prefix Mismatch and Double-Prefixing Breaks Frontend API Calls
- **Severity**: Critical
- **Exact File & Function**: `client/src/api/client.js` (`resolveBaseUrl`), `client/src/api/scans.js` (`uploadScan`), `server/app/main.py`
- **Actual Observed Behavior**:
  `resolveBaseUrl` returns `/api/v1`. `server/app/main.py` previously registered patient CRUD, measurements, `/api/health`, `/api/meta`, and `/api/dashboard/stats` under `/api` directly. Requests to `/patients`, `/dashboard/stats`, `/meta` resulted in `404 Not Found`. Furthermore, `scans.js` called `/v1/patients/${patientId}/scans`, which concatenated to `/api/v1/v1/patients/.../scans` (double prefix).
- **Expected Behavior**: Consistent routing so all frontend API calls reach their intended registered FastAPI endpoints without 404 errors.
- **Root Cause**: Desynchronized migration where sub-routers had `/api/v1/patients` while core endpoints remained under `/api/patients`.
- **Fix Applied**:
  1. Registered patient CRUD, measurement, health, meta, and stats endpoints under both `/api` and `/api/v1` prefixes in `server/app/main.py`.
  2. Normalized `uploadScan` in `client/src/api/scans.js` to `/patients/${patientId}/scans`.
- **Verification**: `test_01_health_and_meta`, `test_02_dashboard_stats`, `test_03_openapi_routes`, `test_04_patient_crud_workflow` all passed in pytest.
- **Status**: **Verified**

### BUG-002: Missing Scan Route Parameter `angle` in Upload
- **Severity**: High
- **Exact File & Function**: `server/app/routes/scans.py` (`upload_scan`), `client/src/api/scans.js` (`uploadScan`)
- **Actual Observed Behavior**:
  The frontend sends `angle` in multipart form data (`'top'` or `'side'`). `upload_scan` in `scans.py` only accepted `file: UploadFile = File(...)`. The scan document in MongoDB stored `"region": patient["injury_region"]`, but completely omitted `"angle"`.
- **Expected Behavior**: Uploaded scans store `"angle": angle` (`'top'` or `'side'`) so photogrammetry views can be distinguished.
- **Root Cause**: The route handler in `scans.py` omitted the `angle` form parameter.
- **Fix Applied**: Added `angle: str = Form(default="top")` to `upload_scan` in `server/app/routes/scans.py` and saved `"angle": normalized_angle` in the MongoDB scan document.
- **Verification**: `test_05_scan_upload_and_angle` passed in pytest; verified `"angle": "side"` stored and retrieved.
- **Status**: **Verified**

### BUG-003: Duplicate Operation ID for Health Endpoint
- **Severity**: Medium
- **Exact File & Function**: `server/app/main.py` (`health`), `server/app/routes/health.py` (`health`)
- **Actual Observed Behavior**:
  Both files registered `@app.get("/api/health") def health():`, causing FastAPI to emit `UserWarning: Duplicate Operation ID health_api_health_get for function health`.
- **Expected Behavior**: Clean route registration with unique operation IDs.
- **Root Cause**: Redundant inline definition in `main.py`.
- **Fix Applied**: Delegated health check exclusively to `server/app/routes/health.py` for both `/api/health` and `/api/v1/health` and removed the duplicate from `main.py`.
- **Verification**: Zero duplicate operation ID warnings emitted during `app.openapi()` inspection.
- **Status**: **Verified**

### BUG-004: Duplicate Prediction Route with Inconsistent Request/Response Behavior
- **Severity**: High
- **Exact File & Function**: `server/app/main.py` (`create_prediction`), `server/app/routes/predictions.py` (`create_prediction`)
- **Actual Observed Behavior**:
  `main.py` defined `POST /api/patients/{id}/predictions` inline with minimal error handling, while `routes/predictions.py` defined `POST /api/v1/patients/{id}/predictions` with `measurement_id` query parameter support and `ApiError` handling.
- **Expected Behavior**: Single consistent prediction endpoint supporting both `/api` and `/api/v1` prefixes.
- **Root Cause**: Legacy inline route was not removed when `predictions.py` was created.
- **Fix Applied**: Removed the duplicate inline handler in `main.py` and decorated `create_prediction` in `predictions.py` with both `/api/v1/patients/{id}/predictions` and `/api/patients/{id}/predictions`.
- **Verification**: `test_06_end_to_end_clinical_pipeline` step 3 passed in pytest.
- **Status**: **Verified**

### BUG-005: Cast Generation API Schema and Path Inconsistency
- **Severity**: Critical
- **Exact File & Function**: `client/src/api/casts.js` (`generateCast`), `client/src/pages/CastDesignerPage.jsx` (`handleGenerateCast`), `server/app/routes/casts.py` (`create_cast`)
- **Actual Observed Behavior**:
  `generateCast` in `casts.js` called `/patients/${patientId}/cast/generate` (which returned 404). `casts.py` expected `CastCreate` with required `prediction_id: str`. `CastDesignerPage.jsx` passed `params` without `prediction_id`, failing with 422.
- **Expected Behavior**: Frontend sends required `prediction_id` and parameter overrides to `POST /api/v1/patients/{patient_id}/casts`. Backend accepts both flat and nested parameter structures.
- **Root Cause**: Incompatible API path and unmapped payload schema.
- **Fix Applied**:
  1. Updated `generateCast` in `client/src/api/casts.js` to call `/patients/${patientId}/casts`.
  2. Captured `predictionId` in `CastDesignerPage.jsx` and passed `{ prediction_id, params, ... }`.
  3. Made `CastCreate` in `casts.py` support nested `params` dictionary fallback and automatic latest prediction lookup if omitted.
- **Verification**: `test_06_end_to_end_clinical_pipeline` step 4 passed in pytest.
- **Status**: **Verified**

### BUG-006: Cast Download URL and Artifact Name Mismatch
- **Severity**: High
